-- Key relationships & constraints
-- 1. Each book belongs to a category (many books → 1 category) ALTER TABLE books ADD FOREIGN KEY (category_id) REFERENCES categories(category_id); -- 2. Each issue record links a book to a member ALTER TABLE issued_books ADD FOREIGN KEY (book_id) REFERENCES books(book_id), ADD FOREIGN KEY (member_id) REFERENCES members(member_id); -- 3. Fine payments linked to issued_books ALTER TABLE fine_payments ADD FOREIGN KEY (issue_id) REFERENCES issued_books(issue_id); -- Stored procedure: issue a book CALL issue_book(book_id, member_id, loan_days); -- Stored procedure: return a book (auto-calculates fine @ ₹5/day) CALL return_book(issue_id); -- View: currently issued with overdue days SELECT * FROM v_issued_books WHERE status = 'issued'; -- Overdue query SELECT * FROM v_issued_books WHERE status = 'issued' AND due_date < CURDATE();
